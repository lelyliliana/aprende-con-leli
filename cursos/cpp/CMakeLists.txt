  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Configuración del curso" FORCE)

  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
function(configurar target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /Zc:__cplusplus)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(CURSO_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()

function(prueba_salida nombre target salida entrada status error)
  add_test(NAME ${nombre} COMMAND "${CMAKE_COMMAND}"
    "-DEXE=$<TARGET_FILE:${target}>" "-DEXPECTED=${CMAKE_CURRENT_SOURCE_DIR}/${salida}"
    "-DINPUT=${entrada}" "-DSTATUS=${status}" "-DERROR_EXPECTED=${error}"
    "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}" "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/ejecuciones/${nombre}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/comprobar.cmake")
endfunction()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/scripts/manifest.json" manifest)
string(JSON n LENGTH "${manifest}")
math(EXPR ultimo "${n}-1")
foreach(i RANGE 0 ${ultimo})
  string(JSON target GET "${manifest}" ${i} target)
  string(JSON fuente GET "${manifest}" ${i} archivo)
  string(JSON salida GET "${manifest}" ${i} salida)
  string(JSON tipo_entrada TYPE "${manifest}" ${i} entrada)
  set(entrada "")
  if(tipo_entrada STREQUAL "STRING")
    string(JSON entrada GET "${manifest}" ${i} entrada)
    set(entrada "${CMAKE_CURRENT_SOURCE_DIR}/${entrada}")
  endif()
  add_executable(${target} "${fuente}")
  if(target STREQUAL "u18_e02")
    target_sources(${target} PRIVATE unidad18-plantillas/ejemplos/operaciones.cpp)
  endif()
  if(target STREQUAL "proyecto_paso1")
    target_link_libraries(${target} PRIVATE inventario_lectura)
  elseif(target STREQUAL "proyecto_paso2" OR target STREQUAL "proyecto_paso3")
    target_link_libraries(${target} PRIVATE inventario_calculo)
  elseif(target MATCHES "^u21_" OR target STREQUAL "proyecto_paso4")
    target_link_libraries(${target} PRIVATE inventario_lib)
  endif()
  configurar(${target})
  prueba_salida(${target} ${target} "${salida}" "${entrada}" 0 "")
endforeach()
